      if (rolUsuarioActual === 'SuperAdmin') return;

          document.getElementById('bannerAlertaPago').style.display = 'none';
          document.getElementById('pantallaBloqueoImpago').style.display = 'none';
          return;
        }
        if (empObj.estado === 'bloqueada') {
          document.getElementById('pantallaBloqueoImpago').style.display = 'flex';
          return;
        }
        if (empObj.fechaPago) {
          var hoy = new Date();
          hoy.setHours(0,0,0,0);
          var fechaPagoObj = new Date(empObj.fechaPago + 'T00:00:00');
          var diffTime = fechaPagoObj - hoy;
          var diasRestantes = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

          if (diasRestantes <= 5 && diasRestantes >= 0) {
            document.getElementById('lblDiasRestantesPago').innerText = diasRestantes;
            document.getElementById('lblFechaVencimientoPago').innerText = fechaPagoObj.toLocaleDateString();

            empObj.estado = 'bloqueada';
            guardarDatosEnNube();
            document.getElementById('pantallaBloqueoImpago').style.display = 'flex';
          }
        }
      }
    }

    async function conectarImpresoraBluetooth() {
      if (!navigator.bluetooth) {
        alert("Tu navegador o dispositivo no soporta Web Bluetooth. Usa Google Chrome para Android o PC.");
        return;
      }
      try {
        document.getElementById('estadoBluetoothLbl').innerText = 'Buscando...';
        document.getElementById('estadoBluetoothLbl').style.color = 'var(--warning)';

        const device = await navigator.bluetooth.requestDevice({
          acceptAllDevices: true,
          optionalServices: ['000018f0-0000-1000-8000-00805f9b34fb', '0000ff00-0000-1000-8000-00805f9b34fb', 'e7810a71-73ae-499d-8c15-faa9aef0c3f2']
        });

        bluetoothDevice = device;
        document.getElementById('nombreImpresoraLbl').innerText = device.name || 'Impresora Bluetooth';
        
        const server = await device.gatt.connect();
        const services = await server.getPrimaryServices();
        
        for (const service of services) {
          const characteristics = await service.getCharacteristics();
          for (const characteristic of characteristics) {
            if (characteristic.properties.write || characteristic.properties.writeWithoutResponse) {
              bluetoothCharacteristic = characteristic;
              break;
            }
          }
          if (bluetoothCharacteristic) break;
        }

        if (bluetoothCharacteristic) {
          document.getElementById('estadoBluetoothLbl').innerText = 'Conectada';
          document.getElementById('estadoBluetoothLbl').style.color = 'var(--success)';
          alert("Impresora Bluetooth conectada con exito!");
        } else {
          alert("Se conecto al dispositivo pero no se encontro canal de escritura compatible.");
        }
      } catch (error) {
        console.error(error);
        document.getElementById('estadoBluetoothLbl').innerText = 'Error / Desconectada';
        document.getElementById('estadoBluetoothLbl').style.color = 'var(--danger)';
        alert("Error al conectar con la impresora: " + error.message);
      }
    }

    function limpiarTextoParaImpresora(texto) {
      if (!texto) return "";
      return texto
        .toString()
        .replace(/á/g, 'a').replace(/é/g, 'e').replace(/í/g, 'i').replace(/ó/g, 'o').replace(/ú/g, 'u')
        .replace(/Á/g, 'A').replace(/É/g, 'E').replace(/Í/g, 'I').replace(/Ó/g, 'O').replace(/Ú/g, 'U')
        .replace(/ñ/g, 'n').replace(/Ñ/g, 'N')
        .replace(/¡/g, '').replace(/¿/g, '');
    }

    async function enviarComandosBluetooth(texto) {
      if (!bluetoothCharacteristic) {
        alert("Primero debes conectar una impresora termica por Bluetooth.");
        abrirModalImpresoraBluetooth();
        return;
      }
      try {
        const initCommand = new Uint8Array([0x1B, 0x40]);
        await bluetoothCharacteristic.writeValue(initCommand);

        const textoLimpio = limpiarTextoParaImpresora(texto);
        const lineas = textoLimpio.split('\n');
        
        const encoder = new TextEncoder();

        for (let i = 0; i < lineas.length; i++) {
          let lineaCompleta = lineas[i] + "\n";
          let data = encoder.encode(lineaCompleta);
          await bluetoothCharacteristic.writeValue(data);
          await new Promise(r => setTimeout(r, 40));
        }

        let avanceFinal = encoder.encode("\n\n\n\n\n");
        await bluetoothCharacteristic.writeValue(avanceFinal);

        alert("Impresion enviada correctamente!");
      } catch (error) {
        console.error(error);
        alert("Error al imprimir por Bluetooth: " + error.message);
      }
    }

    function calcularCuotaPreview() {
      var monto = parseFloat(document.getElementById('pMonto').value) || 0;
      var gastoCierre = parseFloat(document.getElementById('pGastoCierre').value) || 0;
      var tasa = parseFloat(document.getElementById('pInteres').value) || 30;
      var plazo = document.getElementById('pPlazo').value;
      var fechaInicioVal = document.getElementById('pFechaInicio').value;

      var numCuotas = 13;
      var cuota = 0;
      var totalPagar = 0;

      if (plazo === 'diario') {
        numCuotas = 30;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar / numCuotas;
      } else if (plazo === '10_semanas') {
        numCuotas = 10;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar / numCuotas;
      } else if (plazo === 'quincenal') {
        numCuotas = 6;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar / numCuotas;
      } else if (plazo === 'mensual') {
        numCuotas = 3;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar / numCuotas;
      } else if (plazo === 'capital_final') {
        numCuotas = 1;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar;
      } else if (plazo === 'personalizado') {
        numCuotas = parseInt(document.getElementById('pSemanasCustom').value) || 1;
        cuota = parseFloat(document.getElementById('pCuotaCustom').value) || 0;
        totalPagar = cuota * numCuotas;
      } else {
        numCuotas = 13;
        totalPagar = monto + (monto * (tasa / 100));
        cuota = totalPagar / numCuotas;
      }

      var montoEntregar = monto - gastoCierre;

      document.getElementById('prevCuota').innerText = '$' + cuota.toFixed(2);
      document.getElementById('prevTotal').innerText = '$' + totalPagar.toFixed(2);
      document.getElementById('prevNeto').innerText = '$' + montoEntregar.toFixed(2);

      if (fechaInicioVal) {
        var fechaTermino = new Date(fechaInicioVal + 'T00:00:00');
        if (plazo === 'diario') {
          fechaTermino.setDate(fechaTermino.getDate() + numCuotas);
        } else if (plazo === 'quincenal') {
          fechaTermino.setDate(fechaTermino.getDate() + (numCuotas * 15));
        } else if (plazo === 'mensual') {
          fechaTermino.setMonth(fechaTermino.getMonth() + numCuotas);
        } else {
          fechaTermino.setDate(fechaTermino.getDate() + (numCuotas * 7));
        }
        document.getElementById('prevFechaTermino').innerText = fechaTermino.toLocaleDateString();
      } else {
        document.getElementById('prevFechaTermino').innerText = '-';
      }
    }

    function corregirRegistrosExistentes() {
      var modificado = false;
      prestamosDB.forEach(function(p) {
        var cap = parseFloat(p.monto) || 0;
        var tasa = parseFloat(p.tasa) || 30;
        var cuotasNum = parseInt(p.numCuotas) || 13;
        var totalReal = p.plazo === 'personalizado' ? (p.valorCuota * cuotasNum) : (cap + (cap * (tasa / 100)));
        var cuotaReal = p.plazo === 'personalizado' ? p.valorCuota : (totalReal / cuotasNum);

        if (p.valorCuota !== cuotaReal || p.totalPagar !== totalReal) {
          p.valorCuota = cuotaReal;
          p.totalPagar = totalReal;
          modificado = true;
        }
      });
      if (modificado) guardarEnLocalStorageLocal();
    }

    function calcularMontoAtrasoPrestamo(p, hoyObj) {
      if (!p || p.balancePendiente <= 0) return 0;
      if (!hoyObj) {
        hoyObj = new Date();
        hoyObj.setHours(0,0,0,0);
      }
      var pagosP = cobrosDB.filter(function(cp) { return cp.prestamoId == p.id; });
      var totalPagado = pagosP.reduce(function(acc, curr) { return acc + curr.monto; }, 0);

      var fechaInicioObj = new Date(p.fechaInicio + 'T00:00:00');
      var diffDiasInicio = Math.floor((hoyObj - fechaInicioObj) / (1000 * 60 * 60 * 24));
      var diasPorCuota = (p.plazo === 'quincenal') ? 15 : (p.plazo === 'diario' ? 1 : (p.plazo === 'mensual' ? 30 : 7));

      var cuotasEsperadasTranscurridas = Math.floor(diffDiasInicio / diasPorCuota) + 1;
      if (cuotasEsperadasTranscurridas > p.numCuotas) cuotasEsperadasTranscurridas = p.numCuotas;
      if (cuotasEsperadasTranscurridas < 0) cuotasEsperadasTranscurridas = 0;

      var montoDebidoTeorico = cuotasEsperadasTranscurridas * p.valorCuota;
      var montoAtraso = Math.max(0, montoDebidoTeorico - totalPagado);
      
      return Math.min(montoAtraso, p.balancePendiente);
    }

    function realizarLogin(e) {
      e.preventDefault();
      empresaActual = document.getElementById('loginEmpresaSelect').value;
      var rol = document.getElementById('loginRolSelect').value;
      var p = document.getElementById('loginPassword').value.trim();

      if (rol === "SuperAdmin" && (p === "SuperAdmin21" || p === "Admin21")) {
        rolUsuarioActual = "SuperAdmin";
        nombreUsuarioActual = "Super Administrador (Tú)";
        zonaUsuarioActual = "Todas las Zonas (Global)";
        guardarSesionLocal();
      } else if (rol === "Administrador") {
        var empG = empresasGlobalesDB.find(function(item) { return item.nombre === empresaActual; });
        if (empG && (p === empG.passwordAdmin || p === "Admin21")) {
          rolUsuarioActual = "admin";
          nombreUsuarioActual = "Administrador " + empresaActual;
          zonaUsuarioActual = "Todas las Zonas (Oficina)";
          guardarSesionLocal();
        } else {
          alert("Contraseña de Administrador incorrecta para esta empresa.");
        }
      } else if (rol === "cobrador") {
        var nom = document.getElementById('loginNombreSelect').value;
        var zon = document.getElementById('loginZonaSelect').value;
        if(!nom || !zon) return alert("Selecciona el cobrador y la zona correspondiente.");
        var cob = cobradoresDB.find(function(c) { return c.nombre === nom; });
        if(cob && (p === cob.password || p === "Perdomo21")) {
          rolUsuarioActual = "cobrador";
          nombreUsuarioActual = nom;
          zonaUsuarioActual = zon;
          guardarSesionLocal();
        } else {
          alert("Contrasena incorrecta.");
        }
      } else if (rol === "empleado") {
        var nom = document.getElementById('loginNombreSelect').value;
        if(!nom) return alert("Selecciona el empleado.");
        var emp = empleadosDB.find(function(e) { return e.nombre === nom; });
        if(emp && (p === emp.password || p === "Perdomo21")) {
          rolUsuarioActual = "empleado";
          nombreUsuarioActual = nom;
          zonaUsuarioActual = "Zonas Asignadas a sus Cobradores/Rutas";
          guardarSesionLocal();
        } else {
          alert("Contrasena incorrecta.");
        }
      } else {
        alert("Credenciales incorrectas.");
      }
    }

      });

        clientes: [], prestamos: [], cobros: [], notas: [], notificaciones: [],

    function renderSuperAdminEmpresas() {
      var cont = document.getElementById('listaEmpresasAdminDiv');
      if (!cont) return;
      if (empresasGlobalesDB.length === 0) {
        cont.innerHTML = '<p style="font-size:0.85rem; color:#475569;">No hay empresas registradas.</p>';
        return;
      }
      cont.innerHTML = empresasGlobalesDB.map(function(emp) {
        var estadoBadge = emp.estado === 'bloqueada' 
          ? '<span style="background:var(--danger); color:white; padding:2px 6px; border-radius:4px; font-size:0.75rem;">🔒 Bloqueada (Impago)</span>' 
          : '<span style="background:var(--success); color:white; padding:2px 6px; border-radius:4px; font-size:0.75rem;">✅ Activa</span>';
        
        var botonBloqueoToggle = emp.estado === 'bloqueada' 
          ? '<button class="btn btn-success btn-sm" onclick="cambiarEstadoEmpresaAdmin(\'' + String(emp.nombre).replace(/'/g, "\\'") + '\', \'activa\')">🔓 Desbloquear Sistema</button>'
          : '<button class="btn btn-danger btn-sm" onclick="cambiarEstadoEmpresaAdmin(\'' + String(emp.nombre).replace(/'/g, "\\'") + '\', \'bloqueada\')">🔒 Tumbar Sistema</button>';

        var fechaPagoTxt = emp.fechaPago || 'Sin fecha';
        var diasInfo = '';
        if (emp.fechaPago) {
          var hoy = new Date(); hoy.setHours(0,0,0,0);
          var fPago = new Date(emp.fechaPago + 'T00:00:00');
          var dias = Math.ceil((fPago - hoy) / (1000*60*60*24));
          if (dias < 0) diasInfo = ' <span style="color:var(--danger); font-weight:700;">(Vencido hace ' + Math.abs(dias) + ' días)</span>';
          else if (dias <= 5) diasInfo = ' <span style="color:var(--warning); font-weight:700;">(Faltan ' + dias + ' días · Aviso activo)</span>';
          else diasInfo = ' <span style="color:var(--success);">(Faltan ' + dias + ' días)</span>';
        }

        var pagoBadge = (emp.requierePago === false)
          ? '<span style="background:#64748b; color:white; padding:2px 6px; border-radius:4px; font-size:0.7rem;">🆓 Gratis</span>'
          : '<span style="background:var(--warning); color:white; padding:2px 6px; border-radius:4px; font-size:0.7rem;">💳 Paga</span>';
        var logoHtml = emp.logo 
          ? '<img src="' + emp.logo + '" style="width:40px; height:40px; object-fit:contain; border-radius:4px; border:1px solid var(--border); background:white; margin-right:10px;">'
          : '';
        var colorDot = '<span style="display:inline-block; width:14px; height:14px; border-radius:50%; background:' + (emp.colorPrimario || '#1e3a8a') + '; border:1px solid #ccc; vertical-align:middle;"></span>';

        return '<div style="background:#f8fafc; padding:14px; border-radius:8px; margin-top:10px; border:1px solid var(--border); font-size:0.85rem;">' +
          '<div style="display:flex; justify-content:space-between; align-items:flex-start; gap:10px; flex-wrap:wrap;">' +
            '<div style="flex:1; min-width:200px; display:flex; align-items:flex-start;">' +
              logoHtml +
              '<div>' +
                '<strong style="font-size:0.95rem;">' + emp.nombre + '</strong> ' + estadoBadge + ' ' + pagoBadge + '<br>' +
                '<small style="color:#475569; display:block; margin-top:4px;">🆔 RNC: <strong>' + (emp.rnc || 'N/A') + '</strong> · Cédula: <strong>' + (emp.cedula || 'N/A') + '</strong></small>' +
                '<small style="color:#475569; display:block;">👤 Admin: <strong>' + (emp.usuarioAdmin || 'admin') + '</strong> · 🔑 ••••••••</small>' +
                '<small style="color:#475569; display:block;">📅 Fecha de Pago: <strong>' + fechaPagoTxt + '</strong>' + diasInfo + '</small>' +
                '<small style="color:#475569; display:block;">' + colorDot + ' Color de interfaz</small>' +
              '</div>' +
            '</div>' +
            '<div style="display:flex; flex-direction:column; gap:6px; align-items:flex-end;">' +
              '<button class="btn btn-warning btn-sm" onclick="abrirModalEditarEmpresa(' + emp.id + ')" style="background:var(--primary); color:white;">✏️ Editar Datos</button>' +
              botonBloqueoToggle +
            '</div>' +
          '</div>' +
        '</div>';
      }).join('');
    }

    function guardarEdicionEmpresaForm(e) {
      e.preventDefault();
      var id = parseInt(document.getElementById('editEmpGlobalId').value);
      var emp = empresasGlobalesDB.find(function(item) { return item.id == id; });
      if (!emp) return alert("Empresa no encontrada.");

      var nuevoNombre = document.getElementById('editEmpGlobalNombre').value.trim();
      var nuevoRNC = document.getElementById('editEmpGlobalRNC').value.trim();
      var nuevaCedula = document.getElementById('editEmpGlobalCedula').value.trim();
      var nuevoUsuario = document.getElementById('editEmpGlobalUsuario').value.trim();
      var nuevaPass = document.getElementById('editEmpGlobalPassword').value.trim();
      var nuevaFecha = document.getElementById('editEmpGlobalFechaPago').value;
      var nuevoEstado = document.getElementById('editEmpGlobalEstado').value;
      var nuevoRequiere = document.getElementById('editEmpGlobalRequierePago').value === 'true';
      var nuevoColor = document.getElementById('editEmpGlobalColor').value || '#1e3a8a';

      if (!nuevoNombre || !nuevoUsuario || !nuevaPass) {
        alert("Nombre, usuario y contraseña son obligatorios.");
        return;
      }

      var nombreAnterior = emp.nombre;
      emp.nombre = nuevoNombre;
      emp.rnc = nuevoRNC;
      emp.cedula = nuevaCedula;
      emp.usuarioAdmin = nuevoUsuario;
      emp.passwordAdmin = nuevaPass;
      emp.fechaPago = nuevaFecha;
      emp.estado = nuevoEstado;
      emp.requierePago = nuevoRequiere;
      emp.colorPrimario = nuevoColor;
      emp.plan = document.getElementById('editEmpGlobalPlan') ? document.getElementById('editEmpGlobalPlan').value : (emp.plan || 'profesional');
      emp.limiteCobradores = parseInt(document.getElementById('editEmpGlobalLimiteCobradores') ? document.getElementById('editEmpGlobalLimiteCobradores').value : 15) || 15;
      if (logoEmpresaEditTemporal) emp.logo = logoEmpresaEditTemporal;

      if (empresaActual === nombreAnterior && nombreAnterior !== nuevoNombre) {
        empresaActual = nuevoNombre;
        localStorage.setItem('psc_empresa_activa', empresaActual);
      }

      localStorage.setItem('psc_empresas_globales', JSON.stringify(empresasGlobalesDB));
      if (db) db.ref('psc_empresas_globales').set(empresasGlobalesDB);
      guardarDatosEnNube();
      cerrarModalEditarEmpresa();
      alert("✅ Datos de la empresa actualizados.\n\nEstado: " + (nuevoEstado === 'bloqueada' ? '🔒 SISTEMA TUMBADO' : '✅ Activo') + "\nFecha de pago: " + nuevaFecha + (nuevoRNC ? "\nRNC: " + nuevoRNC : ""));
      renderSuperAdminEmpresas();
      actualizarSelectsLoginEmpresa();
      renderEmpresasAisladasList();
      verificarEstadoSuscripcionEmpresa();
      if (empresaActual === emp.nombre) aplicarTemaEmpresa();
    }

    function cargarFiltroZonasClientes() {
      var select = document.getElementById('selectFiltroZonaCliente');
      if (!select) return;
      var valorPrevio = select.value;
      select.innerHTML = '';
      var zonasPermitidas = [];

      if (rolUsuarioActual === 'admin' || rolUsuarioActual === 'SuperAdmin') {
        zonasPermitidas = zonasDB.map(function(z) { return z.nombre; });
      } else if (rolUsuarioActual === 'empleado') {
        var empObj = empleadosDB.find(function(e) { return e.nombre === nombreUsuarioActual; });
        if (empObj) {
          if (empObj.rutasAsignadas) zonasPermitidas = zonasPermitidas.concat(empObj.rutasAsignadas);
          if (empObj.cobradoresAsignados) {
            empObj.cobradoresAsignados.forEach(function(nomCob) {
              var cobObj = cobradoresDB.find(function(co) { return co.nombre === nomCob; });
              if (cobObj && cobObj.zonas) zonasPermitidas = zonasPermitidas.concat(cobObj.zonas);
            });
          }
        }
      } else if (rolUsuarioActual === 'cobrador') {
        zonasPermitidas = [zonaUsuarioActual];
      }

      zonasPermitidas = Array.from(new Set(zonasPermitidas));
      select.innerHTML += '<option value="TODAS">📍 Todas las Zonas Asignadas (' + zonasPermitidas.length + ')</option>';
      zonasPermitidas.forEach(function(z) {
        var objZona = zonasDB.find(function(item) { return item.nombre === z; });
        var numRuta = (objZona && objZona.numero) ? '#' + objZona.numero + ' - ' : '';
        select.innerHTML += '<option value="' + z + '">📍 ' + numRuta + z + '</option>';
      });

      if (valorPrevio && Array.from(select.options).some(function(o) { return o.value === valorPrevio; })) {
        select.value = valorPrevio;
      }
    }

    function iniciarPadFirma() {
      canvasFirma = document.getElementById('canvasFirma');
      if(!canvasFirma) return;
      ctxFirma = canvasFirma.getContext('2d');
      ctxFirma.strokeStyle = "#000000";
      ctxFirma.lineWidth = 2.5;

      function getPos(e) {
        var rect = canvasFirma.getBoundingClientRect();
        var clientX = e.touches ? e.touches[0].clientX : e.clientX;
        var clientY = e.touches ? e.touches[0].clientY : e.clientY;
        return { x: clientX - rect.left, y: clientY - rect.top };
      }

      function startDraw(e) {
        firmando = true;
        var pos = getPos(e);
        ctxFirma.beginPath();
        ctxFirma.moveTo(pos.x, pos.y);
      }

      function moveDraw(e) {
        if(!firmando) return;
        e.preventDefault();
        var pos = getPos(e);
        ctxFirma.lineTo(pos.x, pos.y);
        ctxFirma.stroke();
      }

      function stopDraw() { firmando = false; }

      canvasFirma.addEventListener('mousedown', startDraw);
      canvasFirma.addEventListener('mousemove', moveDraw);
      canvasFirma.addEventListener('mouseup', stopDraw);
      canvasFirma.addEventListener('touchstart', startDraw);
      canvasFirma.addEventListener('touchmove', moveDraw);
      canvasFirma.addEventListener('touchend', stopDraw);
    }

    function limpiarFirma() {
      if(ctxFirma && canvasFirma) {
        ctxFirma.clearRect(0, 0, canvasFirma.width, canvasFirma.height);
      }
    }

    function recalcalcularFechaTerminoEdit() {
      var fechaInicioVal = document.getElementById('editPrestamoFecha').value;
      var plazo = document.getElementById('editPrestamoPlazo').value;
      var numCuotas = (plazo === 'diario') ? 30 : ((plazo === '10_semanas') ? 10 : (plazo === 'quincenal' ? 6 : (plazo === 'mensual' ? 3 : (plazo === 'capital_final' ? 1 : (plazo === 'personalizado' ? (parseInt(document.getElementById('editPlazoSemanasCustom').value) || 1) : 13)))));

      if (fechaInicioVal) {
        var fechaObj = new Date(fechaInicioVal + 'T00:00:00');
        if (plazo === 'diario') {
          fechaObj.setDate(fechaObj.getDate() + numCuotas);
        } else if (plazo === 'quincenal') {
          fechaObj.setDate(fechaObj.getDate() + (numCuotas * 15));
        } else if (plazo === 'mensual') {
          fechaObj.setMonth(fechaObj.getMonth() + numCuotas);
        } else {
          fechaObj.setDate(fechaObj.getDate() + (numCuotas * 7));
        }
        document.getElementById('editPrestamoFechaTermino').value = fechaObj.toLocaleDateString();
      }
    }

      e.preventDefault();

      if(c) {

        c.direccion = c.calle + " #" + c.numCasa + ", " + c.sector + ", " + c.ciudad + " (" + c.provincia + ")";
        c.zona = document.getElementById('editCliZona').value;

        var prestamo = prestamosDB.find(function(p) { return p.clienteId == id && p.balancePendiente > 0; });
        if (prestamo) {
          prestamo.clienteNombre = c.nombre;
          prestamo.cedula = c.cedula;
          prestamo.telefono = c.telefono;
          prestamo.direccion = c.direccion;
          prestamo.monto = parseFloat(document.getElementById('editPrestamoMonto').value) || prestamo.monto;
          prestamo.gastoCierre = parseFloat(document.getElementById('editPrestamoGasto').value) || 0;
          prestamo.montoNeto = prestamo.monto - prestamo.gastoCierre;
          prestamo.fechaInicio = document.getElementById('editPrestamoFecha').value || prestamo.fechaInicio;
          prestamo.plazo = document.getElementById('editPrestamoPlazo').value || prestamo.plazo;

          if (prestamo.plazo === 'diario') {
            prestamo.numCuotas = 30;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar / prestamo.numCuotas;
          } else if (prestamo.plazo === '10_semanas') {
            prestamo.numCuotas = 10;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar / prestamo.numCuotas;
          } else if (prestamo.plazo === 'quincenal') {
            prestamo.numCuotas = 6;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar / prestamo.numCuotas;
          } else if (prestamo.plazo === 'mensual') {
            prestamo.numCuotas = 3;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar / prestamo.numCuotas;
          } else if (prestamo.plazo === 'capital_final') {
            prestamo.numCuotas = 1;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar;
          } else if (prestamo.plazo === 'personalizado') {
            prestamo.numCuotas = parseInt(document.getElementById('editPlazoSemanasCustom').value) || 1;
            prestamo.valorCuota = parseFloat(document.getElementById('editPlazoCuotaCustom').value) || 0;
            prestamo.totalPagar = prestamo.valorCuota * prestamo.numCuotas;
          } else {
            prestamo.numCuotas = 13;
            prestamo.totalPagar = prestamo.monto + (prestamo.monto * ((prestamo.tasa || 30) / 100));
            prestamo.valorCuota = prestamo.totalPagar / prestamo.numCuotas;
          }

          var fechaObj = new Date(prestamo.fechaInicio + 'T00:00:00');
          if (prestamo.plazo === 'diario') {
            fechaObj.setDate(fechaObj.getDate() + prestamo.numCuotas);
          } else if (prestamo.plazo === 'quincenal') {
            fechaObj.setDate(fechaObj.getDate() + (prestamo.numCuotas * 15));
          } else if (prestamo.plazo === 'mensual') {
            fechaObj.setMonth(fechaObj.getMonth() + prestamo.numCuotas);
          } else {
            fechaObj.setDate(fechaObj.getDate() + (prestamo.numCuotas * 7));
          }
          prestamo.fechaTermino = fechaObj.toLocaleDateString();
        }

        guardarDatosEnNube();
        alert('Informacion actualizada correctamente.');
        cerrarPerfilCliente();
      }
    }

    function renderClientes() {
      var contenedor = document.getElementById('listaClientesGeneral');
      var tituloLista = document.getElementById('tituloListaClientes');
      if(!contenedor) return;
      
      var busqueda = document.getElementById('inputBusquedaCliente').value.toLowerCase();
      var filtroZonaSel = document.getElementById('selectFiltroZonaCliente') ? document.getElementById('selectFiltroZonaCliente').value : 'TODAS';
      
      contenedor.innerHTML = '';

      var titulosMap = {
        'todos': '📁 Todos los Clientes',
        'activos': '🟢 Clientes Activos (Al Dia)',
        'atrasados': '🟡 Clientes con Atrasos (14+ dias sin pago)',
        'salientes': '🔵 Clientes Salientes / Renovacion (Semana 10+)',
        'vencidos': '🔴 Clientes Vencidos (Plazo Expirado)'
      };
      if(tituloLista) tituloLista.innerText = titulosMap[filtroClientesActual] || 'Listado de Clientes';

      var hoy = new Date();
      hoy.setHours(0,0,0,0);

      var filtrados = clientesDB.filter(function(c) {
        var apodoText = c.apodo ? c.apodo.toLowerCase() : '';
        var matchTexto = c.nombre.toLowerCase().indexOf(busqueda) !== -1 || apodoText.indexOf(busqueda) !== -1 || c.cedula.indexOf(busqueda) !== -1;
        if (!matchTexto) return false;

        if (filtroZonaSel !== 'TODAS' && c.zona !== filtroZonaSel) return false;

        if (rolUsuarioActual === 'cobrador' && c.zona !== zonaUsuarioActual) return false;

        if (rolUsuarioActual === 'empleado') {
          var empObj = empleadosDB.find(function(e) { return e.nombre === nombreUsuarioActual; });
          if(empObj) {
            var zonasPermitidas = [];
            if (empObj.rutasAsignadas) zonasPermitidas = zonasPermitidas.concat(empObj.rutasAsignadas);
            if (empObj.cobradoresAsignados) {
              empObj.cobradoresAsignados.forEach(function(nomCob) {
                var cobObj = cobradoresDB.find(function(co) { return co.nombre === nomCob; });
                if(cobObj && cobObj.zonas) zonasPermitidas = zonasPermitidas.concat(cobObj.zonas);
              });
            }
            if(!zonasPermitidas.includes(c.zona)) return false;
          }
        }

        if (filtroClientesActual === 'todos') return true;

        var prestamosC = prestamosDB.filter(function(p) { return p.clienteId == c.id && p.balancePendiente > 0; });
        if (prestamosC.length === 0) return false;
        var p = prestamosC[0];
        var pagosP = cobrosDB.filter(function(cp) { return cp.prestamoId == p.id; });
        var totalPagado = pagosP.reduce(function(acc, curr) { return acc + curr.monto; }, 0);
        
        var fechaInicioObj = new Date(p.fechaInicio + 'T00:00:00');
        var diffDiasInicio = Math.floor((hoy - fechaInicioObj) / (1000 * 60 * 60 * 24));
        var cuotasEsperadasTranscurridas = Math.floor(diffDiasInicio / 7) + 1;
        if (cuotasEsperadasTranscurridas > p.numCuotas) cuotasEsperadasTranscurridas = p.numCuotas;

        var cuotasPagadasEfectivas = Math.floor(totalPagado / p.valorCuota);
        var cuotasDebidasReal = cuotasEsperadasTranscurridas - cuotasPagadasEfectivas;
        var diffDiasUltimoPago = pagosP.length > 0 ? Math.floor((hoy - new Date(pagosP[pagosP.length - 1].fecha)) / (1000 * 60 * 60 * 24)) : diffDiasInicio;
        
        var fechaTerminoObj = null;
        if (p.fechaTermino) {
          var partes = p.fechaTermino.split('/');
          if (partes.length === 3) fechaTerminoObj = new Date(partes[2], partes[1] - 1, partes[0]);
          else fechaTerminoObj = new Date(p.fechaTermino);
        }
        var estaVencidoPorFecha = fechaTerminoObj ? (hoy > fechaTerminoObj) : (diffDiasInicio > (p.numCuotas * 7));

        if (filtroClientesActual === 'vencidos') return estaVencidoPorFecha;
        if (filtroClientesActual === 'salientes') return (!estaVencidoPorFecha && (cuotasPagadasEfectivas >= 10 || diffDiasInicio >= 70));
        if (filtroClientesActual === 'atrasados') return (!estaVencidoPorFecha && (diffDiasUltimoPago >= 14 || cuotasDebidasReal > 0));
        if (filtroClientesActual === 'activos') return (!estaVencidoPorFecha && diffDiasUltimoPago < 14 && cuotasDebidasReal <= 0);

        return true;
      });

      if (filtrados.length === 0) {
        contenedor.innerHTML = '<p style="font-size:0.85rem; color:#475569; text-align:center; padding:15px;">No hay clientes en esta seccion para el filtro o zona seleccionada.</p>';
        return;
      }

      contenedor.innerHTML = filtrados.map(function(c) {
        var avatarHtml = c.foto ? '<img src="' + c.foto + '" class="client-avatar">' : '<div class="client-avatar">👤</div>';
        var apodoStr = c.apodo ? ' <span style="color:var(--primary); font-size:0.8rem;">("' + c.apodo + '")</span>' : '';
        return '<div class="client-card" onclick="verPerfilCreditoCliente(' + c.id + ', \'clientes\')">' +
          avatarHtml +
          '<div class="client-info">' +
            '<div class="client-header">' + c.nombre + apodoStr + ' <span style="font-size:0.7rem; color:var(--accent);">[' + (c.zona || 'General') + ']</span></div>' +
            '<div class="client-address">📍 ' + (c.direccion || 'N/A') + '</div>' +
            '<div class="client-phone">📞 ' + c.telefono + ' — Cedula: ' + c.cedula + '</div>' +
          '</div>' +
        '</div>';
      }).join('');
    }

    function calcularRenovacionPreview() {
      var balancePendiente = parseFloat(document.getElementById('renBalancePendiente').value) || 0; 
      var nuevoMontoAprobado = parseFloat(document.getElementById('renNuevoMonto').value) || 0; 
      var gastoCierre = parseFloat(document.getElementById('renGasto').value) || 0; 
      var tasa = parseFloat(document.getElementById('renInteres').value) || 0;
      var plazo = document.getElementById('renPlazo').value;

      var numCuotas = (plazo === 'diario') ? 30 : ((plazo === '10_semanas') ? 10 : (plazo === 'quincenal' ? 6 : (plazo === 'mensual' ? 3 : (plazo === 'capital_final' ? 1 : 13))));
      var netoEntrega = nuevoMontoAprobado - balancePendiente - gastoCierre;
      var totalPagar = nuevoMontoAprobado + (nuevoMontoAprobado * (tasa / 100));
      var valorCuota = totalPagar / numCuotas;

      document.getElementById('renPrevNuevoCap').innerText = '$' + nuevoMontoAprobado.toFixed(2);
      document.getElementById('renPrevNeto').innerText = '$' + (netoEntrega > 0 ? netoEntrega.toFixed(2) : '0.00');

    }

    function procesarRenovacionForm(e) {
      e.preventDefault();
      var pOld = prestamosDB.find(function(item) { return item.id == prestamoActualId; });
      if(!pOld) return;

      var balancePendiente = parseFloat(document.getElementById('renBalancePendiente').value) || 0;
      var nuevoMontoAprobado = parseFloat(document.getElementById('renNuevoMonto').value) || 0;
      var gastoCierre = parseFloat(document.getElementById('renGasto').value) || 0;
      var tasa = parseFloat(document.getElementById('renInteres').value) || 0;
      var plazo = document.getElementById('renPlazo').value;
      var fechaInicio = document.getElementById('renFecha').value;

      if (nuevoMontoAprobado <= (balancePendiente + gastoCierre)) {
        alert("El nuevo prestamo debe ser mayor al balance pendiente mas el gasto de cierre.");
        return;
      }

      var numCuotas = (plazo === 'diario') ? 30 : ((plazo === '10_semanas') ? 10 : (plazo === 'quincenal' ? 6 : (plazo === 'mensual' ? 3 : (plazo === 'capital_final' ? 1 : 13))));
      var totalPagar = nuevoMontoAprobado + (nuevoMontoAprobado * (tasa / 100));
      var valorCuota = totalPagar / numCuotas;
      var netoEntrega = nuevoMontoAprobado - balancePendiente - gastoCierre;

      pOld.balancePendiente = 0;
      pOld.estado = 'Renovado';

      if (balancePendiente > 0) {
        cobrosDB.push({
          id: Date.now() - 1,
          prestamoId: pOld.id,
          clienteNombre: pOld.clienteNombre,
          telefonoCliente: pOld.telefono,
          monto: balancePendiente,
          mora: 0,
          balanceRestante: 0,
          cuotaInfo: "Saldado por Renovacion",
          montoAtraso: 0,
          fecha: new Date().toLocaleDateString(),
          timestamp: Date.now(),
          nota: "Descontado de renovacion",
          tipoPago: "Renovacion",
          gps: 'Oficina Central'
        });
      }

      var clienteObj = clientesDB.find(function(c) { return c.id == pOld.clienteId; });
      var fechaObj = new Date(fechaInicio + 'T00:00:00');
      if (plazo === 'diario') {
        fechaObj.setDate(fechaObj.getDate() + numCuotas);
      } else if (plazo === 'quincenal') {
        fechaObj.setDate(fechaObj.getDate() + (numCuotas * 15));
      } else if (plazo === 'mensual') {
        fechaObj.setMonth(fechaObj.getMonth() + numCuotas);
      } else {
        fechaObj.setDate(fechaObj.getDate() + (numCuotas * 7));
      }
      var fechaTerminoStr = fechaObj.toLocaleDateString();

      var nuevoPrestamo = {
        id: Date.now(),
        clienteId: pOld.clienteId,
        clienteNombre: pOld.clienteNombre,
        cedula: pOld.cedula,
        telefono: pOld.telefono,
        direccion: pOld.direccion,
        monto: nuevoMontoAprobado,
        gastoCierre: gastoCierre,
        montoNeto: netoEntrega,
        tasa: tasa,
        plazo: plazo,
        numCuotas: numCuotas,
        valorCuota: valorCuota,
        totalPagar: totalPagar,
        balancePendiente: totalPagar,
        fechaInicio: fechaInicio,
        fechaTermino: fechaTerminoStr,
        fotoConfirmacion: fotoConfirmacionTemporal,
        firma: clienteObj ? clienteObj.firma : "",
        estado: 'Activo'
      };

      prestamosDB.push(nuevoPrestamo);
      guardarDatosEnNube();
      cerrarModalRenovacion();
      alert("Renovacion procesada con exito!");

    function procesarCobroForm(e) {
      e.preventDefault();
      var monto = parseFloat(document.getElementById('cobroMonto').value) || 0;
      var mora = parseFloat(document.getElementById('cobroMora').value) || 0;
      var tipo = document.getElementById('cobroTipo').value;
      
      var notaFinal = document.getElementById('cobroNotaSelect').value;
      var notaPersonalizada = document.getElementById('cobroNotaInput').value.trim();
      if (notaPersonalizada) notaFinal += ' - ' + notaPersonalizada;

      var prestamo = prestamosDB.find(function(p) { return p.id == prestamoActualId; });
      if (!prestamo) {
        alert("Error: No se encontro el prestamo asociado a este cobro.");
        return;
      }

      var pagosP = cobrosDB.filter(function(cp) { return cp.prestamoId == prestamo.id; });
      var totalPagadoPrevio = pagosP.reduce(function(acc, curr) { return acc + curr.monto; }, 0);
      
      var cuotaInicio = Math.floor(totalPagadoPrevio / prestamo.valorCuota) + 1;
      var numCuotasCUBIERTAS = Math.max(1, Math.round(monto / prestamo.valorCuota));
      var cuotaFin = Math.min(prestamo.numCuotas, cuotaInicio + numCuotasCUBIERTAS - 1);
      
      var cuotaInfoTexto = (cuotaInicio === cuotaFin) ? 
        (cuotaInicio + " de " + prestamo.numCuotas) : 
        (cuotaInicio + " a " + cuotaFin + " de " + prestamo.numCuotas);

      prestamo.balancePendiente -= monto;
      if (prestamo.balancePendiente <= 0) {
        prestamo.balancePendiente = 0;
        prestamo.estado = 'Finalizado';
      }

      var atrasoRestante = calcularMontoAtrasoPrestamo(prestamo);
      var gpsStr = 'Lat: ' + gpsUltimaPosicion.lat + ', Lng: ' + gpsUltimaPosicion.lng;

      var pago = {
        id: Date.now(),
        prestamoId: prestamo.id,
        clienteNombre: prestamo.clienteNombre,
        telefonoCliente: prestamo.telefono,
        monto: monto,
        mora: mora,
        balanceRestante: prestamo.balancePendiente,
        cuotaInfo: cuotaInfoTexto,
        montoAtraso: atrasoRestante,
        fecha: new Date().toLocaleDateString(),
        timestamp: Date.now(),
        nota: notaFinal,
        tipoPago: tipo,
        gps: gpsStr
      };

      cobrosDB.push(pago);
      guardarDatosEnNube();
      cerrarModalCobro();

      mostrarReciboModal(pago);
    }

    function guardarPrestamo(e) {
      e.preventDefault();
      var clienteId = document.getElementById('pClienteSelect').value;
      var cliente = clientesDB.find(function(c) { return c.id == clienteId; });
      if (!cliente) return alert('Selecciona un cliente valido.');

      var monto = parseFloat(document.getElementById('pMonto').value);
      var gastoCierre = parseFloat(document.getElementById('pGastoCierre').value) || 0;
      var tasa = parseFloat(document.getElementById('pInteres').value) || 30;
      var plazo = document.getElementById('pPlazo').value;
      var fechaInicioVal = document.getElementById('pFechaInicio').value;

      var numCuotas = 13;
      var totalPagar = 0;
      var valorCuota = 0;

      if (plazo === 'diario') {
        numCuotas = 30;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar / numCuotas;
      } else if (plazo === '10_semanas') {
        numCuotas = 10;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar / numCuotas;
      } else if (plazo === 'quincenal') {
        numCuotas = 6;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar / numCuotas;
      } else if (plazo === 'mensual') {
        numCuotas = 3;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar / numCuotas;
      } else if (plazo === 'capital_final') {
        numCuotas = 1;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar;
      } else if (plazo === 'personalizado') {
        numCuotas = parseInt(document.getElementById('pSemanasCustom').value) || 1;
        valorCuota = parseFloat(document.getElementById('pCuotaCustom').value) || 0;
        totalPagar = valorCuota * numCuotas;
      } else {
        numCuotas = 13;
        totalPagar = monto + (monto * (tasa / 100));
        valorCuota = totalPagar / numCuotas;
      }

      var montoNetoEntrega = monto - gastoCierre;

      var fechaTermino = new Date(fechaInicioVal + 'T00:00:00');
      if (plazo === 'diario') {
        fechaTermino.setDate(fechaTermino.getDate() + numCuotas);
      } else if (plazo === 'quincenal') {
        fechaTermino.setDate(fechaTermino.getDate() + (numCuotas * 15));
      } else if (plazo === 'mensual') {
        fechaTermino.setMonth(fechaTermino.getMonth() + numCuotas);
      } else {
        fechaTermino.setDate(fechaTermino.getDate() + (numCuotas * 7));
      }
      var fechaTerminoStr = fechaTermino.toLocaleDateString();

      var nuevoPrestamo = {
        id: Date.now(),
        clienteId: cliente.id,
        clienteNombre: cliente.nombre,
        cedula: cliente.cedula,
        telefono: cliente.telefono,
        direccion: cliente.direccion,
        monto: monto,
        gastoCierre: gastoCierre,
        montoNeto: montoNetoEntrega,
        tasa: tasa,
        plazo: plazo,
        numCuotas: numCuotas,
        valorCuota: valorCuota,
        totalPagar: totalPagar,
        balancePendiente: totalPagar,
        fechaInicio: fechaInicioVal,
        fechaTermino: fechaTerminoStr,
        fotoConfirmacion: fotoConfirmacionTemporal,
        firma: cliente.firma || "",
        estado: 'Activo'
      };

      prestamosDB.push(nuevoPrestamo);
      guardarDatosEnNube();
      imprimirContratoDetalle(nuevoPrestamo);
      document.getElementById('formPrestamo').reset();
      document.getElementById('pFechaInicio').valueAsDate = new Date();
      document.getElementById('previewFotoConfirmacion').style.display = 'none';
      fotoConfirmacionTemporal = "";
      calcularCuotaPreview();
    }

    function aprobarBorradoElementoCobrador(notifId, elementoId, tipo) {
      if(confirm("Confirmas aprobar la eliminacion permanente de este registro solicitado por el cobrador (Auditoría Inmutable)?")) {
        if (tipo === 'pago') {
          cobrosDB = cobrosDB.filter(function(cp) { return cp.id !== elementoId; });
        } else if (tipo === 'prestamo') {
          prestamosDB = prestamosDB.filter(function(p) { return p.id !== elementoId; });
          cobrosDB = cobrosDB.filter(function(c) { return c.prestamoId !== elementoId; });
          notasDB = notasDB.filter(function(n) { return n.id !== elementoId; });
        } else if (tipo === 'nota') {
          notasDB = notasDB.filter(function(n) { return n.id !== elementoId; });
        }

        notificacionesDB = notificacionesDB.filter(function(n) { return n.id !== notifId; });
        guardarDatosEnNube();
        alert("Registro eliminado con exito!");
        renderNotificacionesAdmin();
        renderListadoGeneral();
      }
    }

    function actualizarMetricas() {
      var hoyObj = new Date();
      hoyObj.setHours(0,0,0,0);
      var hoyStr = hoyObj.toLocaleDateString();

      var totalActivoCalle = 0;
      var totalVencidoCalle = 0;

      prestamosDB.forEach(function(p) {
        var clienteObj = clientesDB.find(function(c) { return c.id == p.clienteId; });
        var zonaCliente = clienteObj ? clienteObj.zona : '';

        if (rolUsuarioActual === 'cobrador' && zonaCliente !== zonaUsuarioActual) return;

        if (p.balancePendiente > 0) {
          var fechaTerminoObj = null;
          if (p.fechaTermino) {
            var partes = p.fechaTermino.split('/');
            if (partes.length === 3) fechaTerminoObj = new Date(partes[2], partes[1] - 1, partes[0]);
            else fechaTerminoObj = new Date(p.fechaTermino);
          }
          
          var diffDiasInicio = Math.floor((hoyObj - new Date(p.fechaInicio + 'T00:00:00')) / (1000 * 60 * 60 * 24));
          var estaVencido = fechaTerminoObj ? (hoyObj > fechaTerminoObj) : (diffDiasInicio > (p.numCuotas * 7));

          if (estaVencido) {
            totalVencidoCalle += p.balancePendiente;
          } else {
            totalActivoCalle += p.balancePendiente;
          }
        }
      });

      var totalCobrado = cobrosDB.reduce(function(acc, c) { return acc + c.monto; }, 0);
      
      var cuotaSemanalEsperada = prestamosDB.filter(function(p) {
        var clienteObj = clientesDB.find(function(c) { return c.id == p.clienteId; });
        var zonaCliente = clienteObj ? clienteObj.zona : '';
        if (rolUsuarioActual === 'cobrador' && zonaCliente !== zonaUsuarioActual) return false;
        return p.balancePendiente > 0;
      }).reduce(function(acc, p) { return acc + (p.valorCuota || 0); }, 0);

      var cobroEstimadoHoy = prestamosDB.filter(function(p) {
        var clienteObj = clientesDB.find(function(c) { return c.id == p.clienteId; });
        var zonaCliente = clienteObj ? clienteObj.zona : '';
        if (rolUsuarioActual === 'cobrador' && zonaCliente !== zonaUsuarioActual) return false;

        if (p.balancePendiente <= 0) return false;
        var fechaInicioObj = new Date(p.fechaInicio + 'T00:00:00');
        var diaSemanaInicio = fechaInicioObj.getDay();
        var diaSemanaHoy = hoyObj.getDay();
        return diaSemanaInicio === diaSemanaHoy;
      }).reduce(function(acc, p) { return acc + (p.valorCuota || 0); }, 0);

      var cobradoHoyRuta = cobrosDB.filter(function(c) {
        if (c.fecha !== hoyStr) return false;
        if (rolUsuarioActual === 'cobrador') {
          var p = prestamosDB.find(function(item) { return item.id == c.prestamoId; });
          var cli = clientesDB.find(function(item) { return item.id == (p ? p.clienteId : 0); });
          return cli && cli.zona === zonaUsuarioActual;
        }
        return true;
      }).reduce(function(acc, c) { return acc + c.monto; }, 0);

      var ahora = Date.now();
      var sieteDiasMs = 7 * 24 * 60 * 60 * 1000;
      var catorceDiasMs = 14 * 24 * 60 * 60 * 1000;

      var cobrosEstaSemana = cobrosDB.filter(function(c) {
        var t = c.timestamp || c.id;
        return (ahora - t) <= sieteDiasMs;
      }).reduce(function(acc, c) { return acc + c.monto; }, 0);

      var cobrosSemanaAnterior = cobrosDB.filter(function(c) {
        var t = c.timestamp || c.id;
        var diff = ahora - t;
        return diff > sieteDiasMs && diff <= catorceDiasMs;
      }).reduce(function(acc, c) { return acc + c.monto; }, 0);

      document.getElementById('mPrestado').innerText = '$' + totalActivoCalle.toFixed(2);
      document.getElementById('mCapitalVencido').innerText = '$' + totalVencidoCalle.toFixed(2);
      
      var elCobHist = document.getElementById('mCobradoTotal');
      if(elCobHist) elCobHist.innerText = '$' + totalCobrado.toFixed(2);
      
      document.getElementById('mCuotaSemanalEsperada').innerText = '$' + cuotaSemanalEsperada.toFixed(2);
      document.getElementById('mCobroHoy').innerText = '$' + cobroEstimadoHoy.toFixed(2);
      document.getElementById('mCobradoHoyRuta').innerText = '$' + cobradoHoyRuta.toFixed(2);

      var infoAnterior = document.getElementById('infoCobroAnterior');
      if (infoAnterior) {
        var mensajeTendencia = "";
        if (cobrosEstaSemana > cobrosSemanaAnterior) {
          mensajeTendencia = '🚀 <strong>¡El negocio va subiendo!</strong> Has cobrado <strong>+$' + (cobrosEstaSemana - cobrosSemanaAnterior).toFixed(2) + '</strong> más que la semana anterior.';
        } else if (cobrosEstaSemana < cobrosSemanaAnterior) {
          mensajeTendencia = '📉 <strong>El cobro ha bajado.</strong> Has cobrado <strong>-$' + (cobrosSemanaAnterior - cobrosEstaSemana).toFixed(2) + '</strong> menos que la semana pasada.';
        } else {
          mensajeTendencia = '➖ <strong>El rendimiento se mantiene estable</strong> respecto a la semana pasada.';
        }

        infoAnterior.innerHTML = '<div style="line-height: 1.6;">' +
          '📆 <strong>Recaudado Semana Pasada:</strong> $' + cobrosSemanaAnterior.toFixed(2) + '<br>' +
          '📅 <strong>Recaudado Semana Actual:</strong> <span style="color:var(--primary); font-weight:700;">$' + cobrosEstaSemana.toFixed(2) + '</span><br>' +
          '<hr style="margin: 8px 0; border:0; border-top:1px solid var(--border);">' +
          '<div style="font-weight: 700;">' + mensajeTendencia + '</div>' +
        '</div>';
      }
    }
